/* sbn/scalar.h - portable scalar (u64 limb) mpn primitives for libsbn.
 *
 * The scalar-fallback layer: everything the SIMD/u52 stack needs on the u64
 * side that is NOT already covered by the extracted asm kernels
 * (sbn/kernels.h).  All functions are faithful ports of the GMP generic C
 * code (semantics matched exactly: return values, shift-count ranges,
 * overlap rules), with the gmp-impl.h macro plumbing flattened to plain
 * uint64_t.  Header-only, compiles as both C and C++20 (-fno-exceptions
 * -fno-rtti); no dependency on immintrin or gmp headers.
 *
 * Ported from the GNU MP Library 6.3.0 sources:

 *   inline bodies), gmp-impl.h (MPN_INCR_U, MPN_DECR_U, invert_limb,
 *   invert_pi1, udiv_qrnnd_preinv, udiv_qr_3by2), longlong.h
